# Universal Boost Package for Purge of the Crescent Veil engine/kernel dev

# Update APT first
sudo apt-get update -y

# Core build tools
sudo apt-get install -y build-essential cmake ninja-build pkg-config

# Graphics stack (OpenGL, GLFW, GLAD, Wayland/X11 support, SDL3 prep)
sudo apt-get install -y libgl1-mesa-dev libglu1-mesa-dev libglfw3-dev libxrandr-dev libxi-dev libxinerama-dev libxcursor-dev libxkbcommon-dev libwayland-dev wayland-protocols

# Audio stack (OpenAL Soft, PulseAudio, ALSA optional)
sudo apt-get install -y libopenal-dev libpulse-dev libasound2-dev

# Lua 5.4 system bindings (even though you’re building embedded Lua too)
sudo apt-get install -y lua5.4 liblua5.4-dev

# JSON parsing (Codex usually pulls via vcpkg but these cover system level)
sudo apt-get install -y nlohmann-json3-dev

# General utils (Codex safety for network packages and future containers)
sudo apt-get install -y unzip curl git python3 python3-pip python-is-python3

# Safety clean (prevent apt lock issues if rerunning multiple times)
sudo apt-get autoremove -y
